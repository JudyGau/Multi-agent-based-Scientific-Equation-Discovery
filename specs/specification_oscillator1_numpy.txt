"""
Find the mathematical function skeleton that represents acceleration in a damped nonlinear oscillator system with driving force, given data on position, and velocity. 
"""


import numpy as np

#Initialize parameters
MAX_NPARAMS = 10
params = [1.0]*MAX_NPARAMS


@evaluate.run
def evaluate(data: dict) -> float:
    """ Evaluate the equation on data observations."""
    
    # Load data observations
    inputs, outputs = data['inputs'], data['outputs']
    x, v = inputs[:,0], inputs[:,1]
    
    # Optimize parameters based on data
    from scipy.optimize import least_squares
    def residual(params):
        y_pred = equation(x, v, params)
        return y_pred - outputs        # least_squares 的残差向量

    # 参数以**多起点有界最小二乘**求解（与项目评估器 evaluation/problems.py 一致）：
    # 5 个起点取自 U(-1, 1)（逐个裁剪进边界）+ 可选热启动，逐起点取残差平方均值最小者；
    # bounds 防止无界优化让参数发散（NaN/inf），对应 problems.PARAMS_BOUNDS。
    result = least_squares(residual, [1.0]*MAX_NPARAMS, bounds=(-10000.0, 10000.0))
    
    # Return evaluation score
    optimized_params = result.x
    loss = float(np.mean(np.square(result.fun)))

    if not np.isfinite(loss):
        return None
    else:
        return -loss



@equation.evolve
def equation(x: np.ndarray, v: np.ndarray, params: np.ndarray) -> np.ndarray:
    """ Mathematical function for acceleration in a damped nonlinear oscillator

    Args:
        x: A numpy array representing observations of current position.
        v: A numpy array representing observations of velocity.
        params: Array of numeric constants or parameters to be optimized

    Return:
        A numpy array representing acceleration as the result of applying the mathematical function to the inputs.
    """
    dv = params[0] * x  +  params[1] * v  + params[2]
    return dv
